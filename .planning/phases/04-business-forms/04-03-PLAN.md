  <action>
A "deployment" is one equipment assigned to one sampling location for a campaign.
Each deployment creates rows in CampaignEquipment, CampaignSamplingLocation, AND
EquipmentInstallation atomically. This keeps the three tables consistent.

In api/v1/repositories/campaign_repository.py, add:

list_campaign_deployments(conn, campaign_id: int) -> list[dict]:
  SELECT ce.Equipment_ID, e.Identifier AS equipment_identifier,
         csl.SamplingPoint_ID, sp.SamplingPoint AS sampling_point_name,
         ce.Role,
         ei.Installation_ID, ei.InstalledDate
  FROM [dbo].[CampaignEquipment] ce
  JOIN [dbo].[Equipment] e ON e.Equipment_ID = ce.Equipment_ID
  LEFT JOIN [dbo].[CampaignSamplingLocation] csl ON csl.Campaign_ID = ce.Campaign_ID
  LEFT JOIN [dbo].[SamplingPoint] sp ON sp.SamplingPoint_ID = csl.SamplingPoint_ID
  LEFT JOIN [dbo].[EquipmentInstallation] ei
    ON ei.Equipment_ID = ce.Equipment_ID
    AND ei.SamplingPoint_ID = csl.SamplingPoint_ID
    AND ei.Campaign_ID = ce.Campaign_ID
  WHERE ce.Campaign_ID = ?
  Returns [{equipment_id, equipment_identifier, sampling_point_id, sampling_point_name,
            role, installation_id, installed_date}]

create_campaign_deployment(conn, campaign_id: int, equipment_id: int,
                            sampling_point_id: int, role: str | None,
                            notes: str | None) -> int:
  1. Get campaign.CampaignStartDateTime:
     SELECT CampaignStartDateTime FROM [dbo].[Campaign] WHERE Campaign_ID = ?
     If None, raise ValueError("Campaign has no start date — cannot create installation")
  2. INSERT OR IGNORE into CampaignEquipment:
     IF NOT EXISTS (SELECT 1 FROM [dbo].[CampaignEquipment]
                    WHERE Campaign_ID=? AND Equipment_ID=?)
       INSERT INTO [dbo].[CampaignEquipment] (Campaign_ID, Equipment_ID, Role)
       VALUES (?, ?, ?)
  3. INSERT OR IGNORE into CampaignSamplingLocation:
     IF NOT EXISTS (SELECT 1 FROM [dbo].[CampaignSamplingLocation]
                    WHERE Campaign_ID=? AND SamplingPoint_ID=?)
       INSERT INTO [dbo].[CampaignSamplingLocation] (Campaign_ID, SamplingPoint_ID, Role)
       VALUES (?, ?, ?)
  4. INSERT into EquipmentInstallation:
     INSERT INTO [dbo].[EquipmentInstallation]
       (Equipment_ID, SamplingPoint_ID, InstalledDate, Campaign_ID, Notes)
     VALUES (?, ?, ?, ?, ?)
     SELECT @@IDENTITY — returns Installation_ID
  5. conn.commit() — commit all three steps atomically
  Returns Installation_ID (int)

delete_campaign_deployment(conn, campaign_id: int, equipment_id: int,
                            sampling_point_id: int) -> None:
  DELETE FROM [dbo].[EquipmentInstallation]
  WHERE Campaign_ID=? AND Equipment_ID=? AND SamplingPoint_ID=?
  DELETE FROM [dbo].[CampaignEquipment]
  WHERE Campaign_ID=? AND Equipment_ID=?
  DELETE FROM [dbo].[CampaignSamplingLocation]
  WHERE Campaign_ID=? AND SamplingPoint_ID=?
  conn.commit()

In api/v1/endpoints/campaigns.py, add:
  GET  /campaigns/{campaign_id}/deployments
    → list_campaign_deployments, returns list[dict]
  POST /campaigns/{campaign_id}/deployments
    Body: {equipment_id: int, sampling_point_id: int, role: str | None, notes: str | None}
    → create_campaign_deployment, status_code=201
    Returns {installation_id: int}
    If ValueError from repo → HTTPException(400, detail=str(e))
  DELETE /campaigns/{campaign_id}/deployments
    Body: {equipment_id: int, sampling_point_id: int}
    → delete_campaign_deployment, status_code=204

In app/api_client.py (Campaigns section), add:
