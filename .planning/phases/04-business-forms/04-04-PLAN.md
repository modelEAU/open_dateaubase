vector and matrix channels — e.g., a UV spectral axis (200–700nm, 2nm resolution =

Create api/v1/schemas/value_binning.py:
  class ValueBinItem(BaseModel):
    bin_index: int
    lower_bound: float
    upper_bound: float

  class ValueBinningAxisIn(BaseModel):
    name: str
    description: str | None = None
    unit_id: int
    bins: list[ValueBinItem]  # must not be empty

    @field_validator("bins")
    @classmethod
    def bins_not_empty(cls, v):
        if not v: raise ValueError("bins must not be empty")
        return v

  class ValueBinningAxisOut(BaseModel):
    value_binning_axis_id: int
    name: str
    description: str | None
    unit_id: int
    unit_name: str | None
    number_of_bins: int

  class ValueBinningAxisDetail(ValueBinningAxisOut):
    bins: list[ValueBinItem]

Create api/v1/repositories/value_binning_repository.py:

  list_binning_axes(conn) -> list[dict]:
    SELECT vba.ValueBinningAxis_ID, vba.Name, vba.Description,
           vba.NumberOfBins, vba.Unit_ID, u.Unit AS UnitName
    FROM [dbo].[ValueBinningAxis] vba
    LEFT JOIN [dbo].[Unit] u ON u.Unit_ID = vba.Unit_ID
    ORDER BY vba.Name
    Returns [{value_binning_axis_id, name, description, number_of_bins, unit_id, unit_name}]

  get_binning_axis(conn, axis_id: int) -> dict | None:
    Same query with WHERE vba.ValueBinningAxis_ID = ?
    Also fetch bins:
    SELECT ValueBin_ID, BinIndex, LowerBound, UpperBound
    FROM [dbo].[ValueBin] WHERE ValueBinningAxis_ID = ? ORDER BY BinIndex
    Returns dict with 'bins' key added

  insert_binning_axis(conn, data: dict) -> int:
    INSERT INTO [dbo].[ValueBinningAxis] (Name, Description, NumberOfBins, Unit_ID)
    VALUES (?, ?, ?, ?)
    SELECT @@IDENTITY → axis_id
    Then for each bin in data['bins']:
      INSERT INTO [dbo].[ValueBin] (ValueBinningAxis_ID, BinIndex, LowerBound, UpperBound)
      VALUES (?, ?, ?, ?)
    conn.commit()
    Returns axis_id

  delete_binning_axis(conn, axis_id: int) -> None:
    DELETE FROM [dbo].[ValueBin] WHERE ValueBinningAxis_ID = ?
    DELETE FROM [dbo].[ValueBinningAxis] WHERE ValueBinningAxis_ID = ?
    conn.commit()
    Note: do NOT delete if ChannelAxis rows reference this axis — check first:
      SELECT COUNT(*) FROM [dbo].[ChannelAxis] WHERE ValueBinningAxis_ID = ?
      If > 0: raise ValueError("Axis is in use by channels — cannot delete")

  lookup_binning_axes(conn) -> list[dict]:
    SELECT ValueBinningAxis_ID, Name, NumberOfBins FROM [dbo].[ValueBinningAxis]
    Returns [{value_binning_axis_id, name, number_of_bins}]

Create api/v1/endpoints/value_binning.py with APIRouter():
  GET  /value-binning-axes         → list_binning_axes
  GET  /value-binning-axes/{id}    → get_binning_axis (404 if None)
  GET  /value-binning-axes/lookup  → lookup_binning_axes
  POST /value-binning-axes         → insert_binning_axis, body=ValueBinningAxisIn, status_code=201
  DELETE /value-binning-axes/{id}  → delete_binning_axis, status_code=204
                                     ValueError → HTTPException(400, detail=str(e))
