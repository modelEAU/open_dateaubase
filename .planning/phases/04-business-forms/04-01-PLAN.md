Create api/v1/repositories/lookup_repository.py with these functions (all take conn: pyodbc.Connection):

- get_units_lookup(conn) → list[dict]:
  SELECT Unit_ID, Unit FROM [dbo].[Unit] ORDER BY Unit_ID
  Returns [{unit_id, unit}]

- get_laboratories_lookup(conn) → list[dict]:
  SELECT Laboratory_ID, Name FROM [dbo].[Laboratory] ORDER BY Name
  Returns [{laboratory_id, name}]

- get_procedures_lookup(conn) → list[dict]:
  SELECT Procedure_ID, ProcedureName FROM [dbo].[Procedures] ORDER BY ProcedureName
  Returns [{procedure_id, procedure_name}]

- get_samples_lookup(conn) → list[dict]:
  SELECT s.Sample_ID,
         COALESCE(s.Description, 'Sample ' + CAST(s.Sample_ID AS NVARCHAR)) AS Label,
         s.SampleDateTimeStart
  FROM [dbo].[Sample] s ORDER BY s.SampleDateTimeStart DESC
  Returns [{sample_id, label, sample_date}] — most recent first for usability

- get_sampling_points_lookup(conn) → list[dict]:
  SELECT sp.SamplingPoint_ID,
         COALESCE(sp.SamplingPoint, 'Point ' + CAST(sp.SamplingPoint_ID AS NVARCHAR))
           + ' — ' + COALESCE(sp.SamplingLocation, '') AS Label
  FROM [dbo].[SamplingPoint] sp ORDER BY sp.SamplingPoint
  Returns [{sampling_point_id, label}]

- get_equipment_events_lookup(conn) → list[dict]:
  SELECT ee.EquipmentEvent_ID,
         et.EquipmentEventType_Name + ' — ' + e.Identifier
           + ' (' + CONVERT(NVARCHAR, ee.EventDateTimeStart, 120) + ')' AS Label
  FROM [dbo].[EquipmentEvent] ee
  JOIN [dbo].[EquipmentEventType] et ON et.EquipmentEventType_ID = ee.EquipmentEventType_ID
  JOIN [dbo].[Equipment] e ON e.Equipment_ID = ee.Equipment_ID
  ORDER BY ee.EventDateTimeStart DESC
  Returns [{equipment_event_id, label}]
