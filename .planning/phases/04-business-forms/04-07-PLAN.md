<objective>
Extend the ingest API to accept vector and matrix sensor data. A vector observation is
a spectrum or distribution measured at one timestamp — one float per bin of a
ValueBinningAxis. A matrix observation has two axes (e.g., particle size × settling
velocity) — one float per (row_bin, col_bin) pair.

Purpose: Spectral sensors (UV-Vis, fluorescence) and multi-parameter particle counters
produce array-valued data that does not fit in the scalar Value table. This plan builds
the repository functions and new POST /ingest/sensor-vector and /ingest/sensor-matrix
endpoints that write to ValueVector and ValueMatrix tables.
Output: 2 new ingest endpoints, repository functions, updated schemas, api_client additions.
</objective>

insert_vector_values(conn, channel_id: int,
                     binning_axis_id: int,
                     observations: list[dict]) -> int:
  """Insert rows into dbo.ValueVector. Each observation has:
    timestamp: datetime, bin_values: list[float | None], quality_code: int | None
  bin_values[i] maps to ValueBin with BinIndex=i for the given axis.
  """
  First: fetch all ValueBin_IDs for this axis in order:
    SELECT ValueBin_ID, BinIndex FROM [dbo].[ValueBin]
    WHERE ValueBinningAxis_ID = ? ORDER BY BinIndex
    bin_map = {row[1]: row[0] for row in cursor.fetchall()}  # BinIndex → ValueBin_ID

  If len(bin_map) == 0:
    raise ValueError(f"No bins found for axis {binning_axis_id}")

  For each observation in observations:
    For i, value in enumerate(observation["bin_values"]):
      bin_id = bin_map.get(i)
      if bin_id is None: continue  # skip extra values silently
      INSERT INTO [dbo].[ValueVector] (Channel_ID, Timestamp, ValueBin_ID, Value, QualityCode)
      VALUES (?, ?, ?, ?, ?)
      with channel_id, observation["timestamp"], bin_id, value, observation.get("quality_code")
  conn.commit()
  return sum(len(obs["bin_values"]) for obs in observations)  # total rows written

insert_matrix_values(conn, channel_id: int,
                     row_axis_id: int, col_axis_id: int,
                     observations: list[dict]) -> int:
  """Insert rows into dbo.ValueMatrix. Each observation has:
    timestamp: datetime, matrix: list[list[float | None]], quality_code: int | None
  matrix[r][c] maps to (row_bin at row index r, col_bin at col index c).
  """
  Fetch row bins: SELECT ValueBin_ID, BinIndex FROM [dbo].[ValueBin]
    WHERE ValueBinningAxis_ID = ? ORDER BY BinIndex
  Fetch col bins: same query for col_axis_id
  row_map = {idx: vid, ...}; col_map = {idx: vid, ...}

  For each observation:
    For r, row in enumerate(observation["matrix"]):
      For c, value in enumerate(row):
        row_bin_id = row_map.get(r); col_bin_id = col_map.get(c)
        if row_bin_id is None or col_bin_id is None: continue
        INSERT INTO [dbo].[ValueMatrix]
          (Channel_ID, Timestamp, RowValueBin_ID, ColValueBin_ID, Value, QualityCode)
        VALUES (?, ?, ?, ?, ?, ?)
  conn.commit()
  return total cells written

In api/v1/endpoints/ingest.py, add two new routes:

  @router.post("/sensor-vector", response_model=IngestResponse, status_code=201)
  def ingest_sensor_vector(data: VectorSensorIngestRequest, conn=Depends(get_db)):
    channel_id = find_or_create_sensor_metadata(
        conn, equipment_id=data.equipment_id, parameter_id=data.parameter_id,
        unit_id=data.unit_id, data_provenance_id=data.data_provenance_id,
        processing_degree_id=data.processing_degree_id, value_type_id=2)
    upsert_channel_axis(conn, channel_id, axis_role=0,
                        binning_axis_id=data.binning_axis_id)
    observations = [o.model_dump() for o in data.observations]
    rows = insert_vector_values(conn, channel_id, data.binning_axis_id, observations)
    return IngestResponse(channel_id=channel_id, rows_written=rows)

  @router.post("/sensor-matrix", response_model=IngestResponse, status_code=201)
  def ingest_sensor_matrix(data: MatrixSensorIngestRequest, conn=Depends(get_db)):
    channel_id = find_or_create_sensor_metadata(
        conn, equipment_id=data.equipment_id, parameter_id=data.parameter_id,
        unit_id=data.unit_id, data_provenance_id=data.data_provenance_id,
        processing_degree_id=data.processing_degree_id, value_type_id=3)
    upsert_channel_axis(conn, channel_id, axis_role=0,
                        binning_axis_id=data.row_axis_id)
    upsert_channel_axis(conn, channel_id, axis_role=1,
                        binning_axis_id=data.col_axis_id)
    observations = [o.model_dump() for o in data.observations]
    rows = insert_matrix_values(conn, channel_id, data.row_axis_id,
                                data.col_axis_id, observations)
    return IngestResponse(channel_id=channel_id, rows_written=rows)
