"""Integration tests for v2.2.0 Observation migration schema.

Verifies:
- Observation table exists with correct columns and constraints
- Value, ValueVector, ValueMatrix, ValueImage are restructured (no Channel_ID/Timestamp)
- Annotation has nullable Observation_ID column
- Round-trip: insert Observation + scalar Value, then query via get_scalar_values
"""
import pytest
from .conftest import column_exists, get_table_names

pytestmark = pytest.mark.db

V220_NEW_TABLES = {"Observation"}

V220_REMOVED_COLUMNS = {
    "Value":       {"Value_ID", "Channel_ID", "Timestamp"},
    "ValueVector": {"Channel_ID", "Timestamp"},
    "ValueMatrix": {"Channel_ID", "Timestamp"},
    "ValueImage":  {"ValueImage_ID", "Channel_ID", "Timestamp"},
}

V220_ADDED_COLUMNS = {
    "Observation": {"Observation_ID", "Channel_ID", "Timestamp", "DataType"},
    "ValueImage":  {"Observation_ID"},
    "Annotation":  {"Observation_ID"},
}


class TestV220Schema:
    def test_observation_table_exists(self, db_conn):
        tables = get_table_names(db_conn)
        assert "Observation" in tables

    def test_observation_columns(self, db_conn):
        for col in V220_ADDED_COLUMNS["Observation"]:
            assert column_exists(db_conn, "Observation", col), \
                f"Observation.{col} missing"

    def test_value_dropped_columns(self, db_conn):
        for col in V220_REMOVED_COLUMNS["Value"]:
            assert not column_exists(db_conn, "Value", col), \
                f"Value.{col} should have been dropped"

    def test_value_has_observation_id(self, db_conn):
        assert column_exists(db_conn, "Value", "Observation_ID")

    def test_valuevector_dropped_columns(self, db_conn):
        for col in V220_REMOVED_COLUMNS["ValueVector"]:
            assert not column_exists(db_conn, "ValueVector", col), \
                f"ValueVector.{col} should have been dropped"

    def test_valuematrix_dropped_columns(self, db_conn):
        for col in V220_REMOVED_COLUMNS["ValueMatrix"]:
            assert not column_exists(db_conn, "ValueMatrix", col)

    def test_valueimage_swapped_pk(self, db_conn):
        assert not column_exists(db_conn, "ValueImage", "ValueImage_ID")
        assert column_exists(db_conn, "ValueImage", "Observation_ID")

    def test_annotation_observation_id_nullable(self, db_conn):
        """Annotation.Observation_ID must exist and be nullable."""
        assert column_exists(db_conn, "Annotation", "Observation_ID")
        cursor = db_conn.cursor()
        cursor.execute("""
            SELECT is_nullable FROM INFORMATION_SCHEMA.COLUMNS
            WHERE TABLE_NAME = 'Annotation' AND COLUMN_NAME = 'Observation_ID'
        """)
        row = cursor.fetchone()
        assert row is not None
        assert row[0] == 'YES', "Annotation.Observation_ID must be nullable"

    def test_processing_lineage_unchanged(self, db_conn):
        """ProcessingLineage must NOT have Observation_ID (left intentionally unchanged)."""
        assert not column_exists(db_conn, "ProcessingLineage", "Observation_ID")

class TestObservationAwareIngest:
    """Contract tests verifying ingest endpoints work after Observation refactor.

    These tests mock the repository layer to confirm:
    1. insert_scalar_values is called (via the endpoint's internal flow)
    2. The endpoint returns expected response shape
    3. No KeyError or AttributeError from stale column references
    """

    def test_post_ingest_sensor_scalar_contract(self, client):
        """POST /ingest/sensor returns {"rows_written": N} with valid input."""
        from unittest.mock import patch, MagicMock

        # Patch the repository functions the endpoint calls
        with patch("api.v1.repositories.ingestion_repository.find_or_create_sensor_metadata",
                   return_value=42) as mock_channel, \
             patch("api.v1.repositories.value_repository.insert_scalar_values",
                   return_value=2) as mock_insert:
            resp = client.post("/api/v1/ingest/sensor", json={
                "equipment_id": 1,
                "parameter_id": 1,
                "unit_id": 1,
                "data_provenance_id": 1,
                "processing_degree_id": 1,
                "values": [
                    {"timestamp": "2024-01-01T10:00:00", "value": 7.2},
                    {"timestamp": "2024-01-01T10:05:00", "value": 7.3},
                ]
            })
        assert resp.status_code == 200
        body = resp.json()
        assert "rows_written" in body
        assert body["rows_written"] == 2

        """GET /ingest/last-timestamp returns {"last_timestamp": ...} or null."""

                "equipment_id": 1,
                "parameter_id": 1,
                "data_provenance_id": 1,
                "processing_degree_id": 1,
            })
        assert resp.status_code == 200
        body = resp.json()
        assert "last_timestamp" in body
